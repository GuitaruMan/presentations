# 2028 과목 선택 안내 — DB 이원화 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 정적 JSON 8개 파일에 흩어진 대학 권장과목·학교 교육과정 데이터를 Supabase(PostgreSQL)로 옮기고, 학생 화면(app.js)은 그대로 유지한 채 데이터 소스만 교체한다. 관리자(admin.html)는 로그인 후 개설현황과 학교 교육과정을 직접 DB에 편집할 수 있게 한다.

**Architecture:** 기존 `build_*.py` 빌더 파이프라인(PDF/엑셀 → JSON)은 그대로 두고 마지막 저장 단계만 Supabase upsert로 바꾼다. app.js는 `getJSON()` 호출부만 Supabase REST 호출로 바꾸고, 렌더링 로직은 무수정으로 둔다(같은 모양의 JS 객체를 조립해서 넘긴다). admin.html은 Supabase Auth(이메일+비밀번호, 관리자 1명)로 로그인하고, 개설현황 편집과 신설 학교 교육과정 폼(개설과목 표 + 이수규칙 빌더) 둘 다 Supabase 클라이언트로 직접 쓴다.

**Tech Stack:** Supabase(PostgreSQL, Auth, 자동 REST API), Vanilla JS(기존 스택 유지, 프레임워크 없음), `@supabase/supabase-js`(CDN `<script>` 로드, 빌드 도구 없음), Python(기존 빌더 스크립트, `supabase-py` 라이브러리 추가)

**Spec:** `docs/superpowers/specs/2026-09-01-2028과목선택-db전환-design.md`

## Global Constraints

- 학생 조회 화면은 로그인 없이 Supabase anon key로만 읽는다 — 쓰기 권한은 관리자 세션에만 부여한다(Row Level Security로 강제).
- 학생 개인 선택(내 과목 담기)은 서버에 저장하지 않는다 — localStorage 유지, 이번 계획에서 손대지 않는다.
- 관리자는 1명뿐이다 — 회원가입 UI를 만들지 않는다. 계정은 Supabase 대시보드에서 수동 생성한다.
- 기존 `build_*.py`의 PDF/엑셀 판독 로직(정규식, 표 파싱)은 건드리지 않는다 — 손대는 것은 각 스크립트의 마지막 저장 함수뿐이다.
- app.js의 렌더링 함수(`cardHTML`, `subRows`, `조건에맞나` 등)는 무수정 — 데이터 로딩부(`loadCommon`, `loadCohort`, `getJSON`)만 교체한다.
- 마이그레이션 중에도 기존 정적 JSON은 삭제하지 않는다(태스크 7까지는 백업 용도로 유지).
- 캐시 무효화 관례를 유지한다 — app.js를 수정할 때마다 `VERSION` 변수와 `index.html`의 `?v=` 쿼리를 함께 올린다.

-- supabase_schema.sql
-- 과목 마스터
create table subjects (
  id bigint generated always as identity primary key,
  name text not null unique,
  교과군 text not null,
  유형 text not null,
  별칭 jsonb not null default '[]',
  설명 text,
  출처쪽 integer
);

-- 대학
create table universities (
  id bigint generated always as identity primary key,
  이름 text not null unique,
  출처 text,
  발표일 text,
  제시범위 text,
  제시강도 text,
  강도근거 text,
  안내 jsonb not null default '[]'
);

-- 모집단위
create table admission_units (
  id bigint generated always as identity primary key,
  university_id bigint not null references universities(id) on delete cascade,
  계열 text,
  단과대학 text,
  이름 text not null,
  핵심 jsonb not null default '[]',
  권장 jsonb not null default '[]',
  조건 jsonb not null default '[]',
  비고 text default '',
  권역 text,
  지역 text
);
create index on admission_units(university_id);

-- 코호트 (2025-g3, 2026-g2 등)
create table school_cohorts (
  id bigint generated always as identity primary key,
  코호트id text not null unique,
  입학연도 integer not null,
  대상 text,
  제목 text,
  설명 text,
  선택군 jsonb not null default '[]',
  노출 boolean not null default false
);

-- 코호트별 개설과목
create table school_offerings (
  id bigint generated always as identity primary key,
  cohort_id bigint not null references school_cohorts(id) on delete cascade,
  과목 text not null,
  교과군 text not null,
  유형 text not null,
  운영학점 integer not null,
  선택군 text not null,
  학년 integer not null,
  학기 jsonb not null default '{}',
  평가 text not null
);
create index on school_offerings(cohort_id);

-- 코호트별 이수규칙 전체 (한 코호트당 1행)
create table school_rules (
  id bigint generated always as identity primary key,
  cohort_id bigint not null unique references school_cohorts(id) on delete cascade,
  이수규칙 jsonb not null default '{}',
  선수과목 jsonb not null default '{}',
  선택슬롯 jsonb not null default '{}',
  유의사항 jsonb not null default '[]',
  과정구분 jsonb not null default '[]'
);

-- 코호트별 개설현황(폐강)
create table closed_subjects (
  id bigint generated always as identity primary key,
  cohort_id bigint not null references school_cohorts(id) on delete cascade,
  과목 text not null,
  학기 jsonb not null default '[]',
  사유 text,
  갱신일 text
);
create index on closed_subjects(cohort_id);

-- Row Level Security: 전부 활성화
alter table subjects enable row level security;
alter table universities enable row level security;
alter table admission_units enable row level security;
alter table school_cohorts enable row level security;
alter table school_offerings enable row level security;
alter table school_rules enable row level security;
alter table closed_subjects enable row level security;

-- 읽기: 누구나(anon 포함) 허용
create policy "public read" on subjects for select using (true);
create policy "public read" on universities for select using (true);
create policy "public read" on admission_units for select using (true);
create policy "public read" on school_cohorts for select using (true);
create policy "public read" on school_offerings for select using (true);
create policy "public read" on school_rules for select using (true);
create policy "public read" on closed_subjects for select using (true);

-- 쓰기: 로그인한 사용자만 허용 (관리자 1명)
create policy "auth write" on subjects for all using (auth.role() = 'authenticated');
create policy "auth write" on universities for all using (auth.role() = 'authenticated');
create policy "auth write" on admission_units for all using (auth.role() = 'authenticated');
create policy "auth write" on school_cohorts for all using (auth.role() = 'authenticated');
create policy "auth write" on school_offerings for all using (auth.role() = 'authenticated');

```python
# -*- coding: utf-8 -*-
"""기존 recommendations.json 등 8개 파일을 Supabase 테이블에 적재한다.
   여러 번 실행해도 안전하도록(idempotent) 시작 전에 대상 테이블을 비운다.
   순서: subjects → universities → admission_units → school_cohorts
        → school_offerings → school_rules → closed_subjects
"""
import io
import json
import os

from dotenv import load_dotenv
from supabase import create_client

HERE = os.path.dirname(os.path.abspath(__file__))
ROOT = os.path.dirname(os.path.dirname(HERE))
DATA = os.path.join(ROOT, "ForWeb", "work", "교육과정", "2028과목선택", "data")

load_dotenv(os.path.join(HERE, ".env"))
sb = create_client(os.environ["SUPABASE_URL"], os.environ["SUPABASE_SERVICE_KEY"])


def load(name):
    return json.load(io.open(os.path.join(DATA, name), encoding="utf-8"))


def migrate_subjects():
    rows = load("subject_master.json")
    sb.table("subjects").delete().neq("id", 0).execute()
    payload = [{
        "name": r["name"], "교과군": r["교과군"], "유형": r["유형"],
        "별칭": r.get("별칭", []), "설명": r.get("설명", ""),
        "출처쪽": r.get("출처쪽"),
    } for r in rows]
    sb.table("subjects").insert(payload).execute()
    print("subjects:", len(payload))


def migrate_universities_and_units():
    rec = load("recommendations.json")
    sb.table("admission_units").delete().neq("id", 0).execute()
    sb.table("universities").delete().neq("id", 0).execute()

    uni_total, unit_total = 0, 0
    for u in rec["대학"]:
        res = sb.table("universities").insert({
            "이름": u["대학"], "출처": u.get("출처", ""),
            "발표일": u.get("발표일", ""), "제시범위": u.get("제시범위", ""),
            "제시강도": u.get("제시강도", ""), "강도근거": u.get("강도근거", ""),
            "안내": u.get("안내", []),
        }).execute()
        uni_id = res.data[0]["id"]
        uni_total += 1

        units_payload = [{
            "university_id": uni_id, "계열": m.get("계열", ""),
            "단과대학": m.get("단과대학", ""), "이름": m["모집단위"],
            "핵심": m.get("핵심", []), "권장": m.get("권장", []),
            "조건": m.get("조건", []), "비고": m.get("비고", ""),
            "권역": m.get("권역"), "지역": m.get("지역"),
        } for m in u["모집단위"]]
        if units_payload:
            sb.table("admission_units").insert(units_payload).execute()
            unit_total += len(units_payload)
    print("universities:", uni_total, "/ admission_units:", unit_total)


def migrate_school_cohorts():
    cohorts = load("cohorts.json")
    sb.table("school_cohorts").delete().neq("id", 0).execute()
    id_map = {}
    for c in cohorts["목록"]:
        res = sb.table("school_cohorts").insert({
            "코호트id": c["id"], "입학연도": c["입학연도"], "대상": c.get("대상", ""),
            "제목": c.get("제목", ""), "설명": c.get("설명", ""),
            "선택군": c.get("선택군", []), "노출": c.get("노출", False),
        }).execute()
        id_map[c["id"]] = {"db_id": res.data[0]["id"],
                            "편성표": c["편성표"], "폐강": c["폐강"]}
    print("school_cohorts:", len(id_map))
    return id_map


def migrate_school_offerings_and_rules(id_map):
    sb.table("school_rules").delete().neq("id", 0).execute()
    sb.table("school_offerings").delete().neq("id", 0).execute()

    # 같은 편성표 파일을 공유하는 코호트가 있으므로 파일명 기준으로 캐시한다.
    school_cache = {}
    off_total = 0
    for cid, info in id_map.items():
        fname = info["편성표"]
        if fname not in school_cache:
            school_cache[fname] = load(fname)
        school = school_cache[fname]

        offerings_payload = [{
            "cohort_id": info["db_id"], "과목": r["과목"], "교과군": r["교과군"],
            "유형": r["유형"], "운영학점": r["운영학점"], "선택군": r["선택군"],
            "학년": r["학년"], "학기": r.get("학기", {}), "평가": r["평가"],
        } for r in school["개설"]]
        if offerings_payload:
            sb.table("school_offerings").insert(offerings_payload).execute()
            off_total += len(offerings_payload)

        meta = school["meta"]
        sb.table("school_rules").insert({
            "cohort_id": info["db_id"],
            "이수규칙": meta.get("이수규칙", {}),
            "선수과목": meta.get("선수과목", {}),
            "선택슬롯": meta.get("선택슬롯", {}),
            "유의사항": meta.get("유의사항", []),
            "과정구분": meta.get("과정구분", []),
        }).execute()
    print("school_offerings:", off_total, "/ school_rules:", len(id_map))


def migrate_closed(id_map):
    sb.table("closed_subjects").delete().neq("id", 0).execute()
    total = 0
    for cid, info in id_map.items():
        try:
            closed = load(info["폐강"])
        except FileNotFoundError:
            continue
        payload = [{
            "cohort_id": info["db_id"], "과목": x["과목"],
            "학기": x.get("학기", []), "사유": x.get("사유", ""),
            "갱신일": x.get("갱신일", ""),
        } for x in closed.get("폐강", [])]
        if payload:
            sb.table("closed_subjects").insert(payload).execute()
            total += len(payload)
    print("closed_subjects:", total)


if __name__ == "__main__":
    migrate_subjects()
    migrate_universities_and_units()
    id_map = migrate_school_cohorts()
    migrate_school_offerings_and_rules(id_map)
    migrate_closed(id_map)
    print("이관 완료")
```

"""빌더가 만든 대학 하나 분량의 모집단위 리스트를 Supabase에 반영한다.
   기존 build_*.py 는 recommendations.json 을 직접 갱신했지만,
   이제 이 함수 하나를 호출하는 것으로 대체한다.
"""
import os
from dotenv import load_dotenv
from supabase import create_client

HERE = os.path.dirname(os.path.abspath(__file__))
load_dotenv(os.path.join(HERE, ".env"))
_sb = None


def sb():
    global _sb
    if _sb is None:
        _sb = create_client(os.environ["SUPABASE_URL"], os.environ["SUPABASE_SERVICE_KEY"])
    return _sb


def sync_university(대학이름, units, 출처=None, 발표일=None, 제시범위=None,
                     제시강도=None, 강도근거=None, 안내=None):
    """units 는 build_snu.py/build_korea.py 가 만드는 것과 같은 모양의
       딕셔너리 리스트 — 모집단위/계열/단과대학/핵심/권장/조건/비고/권역/지역 키를 갖는다."""
    existing = sb().table("universities").select("id").eq("이름", 대학이름).execute()
    if existing.data:
        uni_id = existing.data[0]["id"]
        update = {}
        if 출처 is not None: update["출처"] = 출처
        if 발표일 is not None: update["발표일"] = 발표일
        if 제시범위 is not None: update["제시범위"] = 제시범위
        if 제시강도 is not None: update["제시강도"] = 제시강도
        if 강도근거 is not None: update["강도근거"] = 강도근거
        if 안내 is not None: update["안내"] = 안내
        if update:
            sb().table("universities").update(update).eq("id", uni_id).execute()
    else:
        res = sb().table("universities").insert({
            "이름": 대학이름, "출처": 출처 or "", "발표일": 발표일 or "",
            "제시범위": 제시범위 or "", "제시강도": 제시강도 or "",
            "강도근거": 강도근거 or "", "안내": 안내 or [],
        }).execute()
        uni_id = res.data[0]["id"]

    before = sb().table("admission_units").select("id", count="exact").eq("university_id", uni_id).execute()
    sb().table("admission_units").delete().eq("university_id", uni_id).execute()

    payload = [{
        "university_id": uni_id, "계열": u.get("계열", ""), "단과대학": u.get("단과대학", ""),
        "이름": u["모집단위"], "핵심": u.get("핵심", []), "권장": u.get("권장", []),
        "조건": u.get("조건", []), "비고": u.get("비고", ""),
        "권역": u.get("권역"), "지역": u.get("지역"),
    } for u in units]
    if payload:
        sb().table("admission_units").insert(payload).execute()

    print(대학이름, "모집단위", before.count, "->", len(payload))
    return len(payload)
